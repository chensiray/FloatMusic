cmake_minimum_required(VERSION 3.24)
project(FloatMusic VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Multimedia Concurrent)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(FloatMusic src/main.cpp src/playercontroller.cpp src/playercontroller.h src/audiofilepolicy.cpp src/audiofilepolicy.h)
qt_add_qml_module(FloatMusic URI FloatMusic QML_FILES qml/Main.qml)
target_link_libraries(FloatMusic PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Concurrent)
set_target_properties(FloatMusic PROPERTIES WIN32_EXECUTABLE TRUE QT_QML_ROOT_PATH "${CMAKE_CURRENT_SOURCE_DIR}/qml")
if(ANDROID)
    set_target_properties(FloatMusic PROPERTIES
        QT_ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android"
        QT_ANDROID_MIN_SDK_VERSION 28
        QT_ANDROID_TARGET_SDK_VERSION 36
        QT_ANDROID_VERSION_CODE 2
        QT_ANDROID_VERSION_NAME "0.2.0"
        QT_ANDROID_APP_NAME "浮音 · 功能验证")
endif()
option(FLOATMUSIC_TESTS "Build import policy tests (desktop only)" OFF)
if(FLOATMUSIC_TESTS AND NOT ANDROID)
    enable_testing()
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(policy_tests tests/policy_tests.cpp src/audiofilepolicy.cpp src/audiofilepolicy.h)
    target_include_directories(policy_tests PRIVATE src)
    target_link_libraries(policy_tests PRIVATE Qt6::Test)
    add_test(NAME import_policy COMMAND policy_tests)
    qt_add_executable(player_tests tests/player_tests.cpp src/playercontroller.cpp src/playercontroller.h src/audiofilepolicy.cpp src/audiofilepolicy.h)
    target_include_directories(player_tests PRIVATE src)
    target_compile_definitions(player_tests PRIVATE FLOATMUSIC_QML_FILE="${CMAKE_CURRENT_SOURCE_DIR}/qml/Main.qml")
    target_link_libraries(player_tests PRIVATE Qt6::Test Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Concurrent)
    add_test(NAME player_integration COMMAND player_tests -platform offscreen)
    set_target_properties(policy_tests player_tests PROPERTIES WIN32_EXECUTABLE FALSE)
    set_tests_properties(player_integration PROPERTIES ENVIRONMENT "QT_QUICK_BACKEND=software")
endif()
